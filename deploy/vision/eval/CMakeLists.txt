cmake_minimum_required(VERSION 3.16)
project(assistant_vision_eval)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(yolo_ros REQUIRED)
find_package(yolo_onnxruntime_vendor REQUIRED)
find_package(OpenCV REQUIRED)
find_package(cv_bridge REQUIRED)
add_executable(vision-eval runner.cpp)
target_include_directories(vision-eval PRIVATE ${yolo_onnxruntime_vendor_INCLUDE_DIRS})
ament_target_dependencies(vision-eval yolo_ros)
target_link_libraries(vision-eval ${OpenCV_LIBS} cv_bridge::cv_bridge)
